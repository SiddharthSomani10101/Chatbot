from langsmith import traceable
from datetime import datetime
import hashlib
import json

# Simple in-memory state (replace with Redis/DB in prod)
APP_STATE = {
    "summary": "",
    "facts": {},              # e.g. {"employee_count": 5, "shift_b_id": 9}
    "last_intent_hash": None,
    "last_result": None,
    "updated_at": None,
}

def hash_intent(user_text: str) -> str:
    return hashlib.sha256(user_text.strip().lower().encode()).hexdigest()

def build_compact_messages(full_messages, summary, k=4):
    # keep only last k messages + one system summary
    recent = full_messages[-k:] if len(full_messages) > k else full_messages
    compact = []
    if summary:
        compact.append({
            "role": "system",
            "content": f"Conversation summary/state:\n{summary}"
        })
    compact.extend(recent)
    return compact

@traceable(run_type="chain", name="run_agent", project_name="first")
def run_agent(agent, full_messages):
    user_text = full_messages[-1]["content"] if full_messages else ""
    intent_hash = hash_intent(user_text)

    # 1) Idempotency guard: same intent repeated -> return previous result
    if APP_STATE["last_intent_hash"] == intent_hash and APP_STATE["last_result"] is not None:
        return {
            "messages": full_messages + [{
                "role": "assistant",
                "content": f"(Reused previous result)\n{APP_STATE['last_result']}"
            }],
            "reused": True
        }

    # 2) Send compact context instead of full chat history
    compact_messages = build_compact_messages(
        full_messages,
        summary=APP_STATE["summary"],
        k=4
    )

    result = agent.invoke({"messages": compact_messages})

    # 3) Update rolling state (store only key facts + short summary)
    final_text = ""
    if isinstance(result, dict) and "messages" in result and result["messages"]:
        last = result["messages"][-1]
        final_text = last.get("content", "") if isinstance(last, dict) else str(last)
    else:
        final_text = str(result)

    APP_STATE["last_intent_hash"] = intent_hash
    APP_STATE["last_result"] = final_text
    APP_STATE["updated_at"] = datetime.utcnow().isoformat() + "Z"

    # lightweight summary update (replace with better summarizer if needed)
    APP_STATE["summary"] = (
        f"Last user intent: {user_text}\n"
        f"Last assistant response: {final_text[:400]}"
    )

    return result
